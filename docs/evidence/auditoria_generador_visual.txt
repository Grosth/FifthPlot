import json
import gc
import time
from pathlib import Path

import torch
import diffusers
from diffusers import StableDiffusionPipeline

MODEL_DIR = Path("models/stable-diffusion-v1-5")
OUTPUT_DIR = Path("outputs/images")

# Configuracion de la primera prueba.
SEED = 42
IMAGE_SIZE = 512
STEPS = 15
GUIDANCE = 7.0

PROMPT = """Watercolor storybook art of Luma, a small red orange fox with golden eyes, tiny nose, fluffy messy fur and a large curled comet tail. Full body, all paws visible, standing in a teal enchanted forest. Soft moonlight, painterly gouache, visible brushstrokes, warm fur and cool blue green background."""

NEGATIVE_PROMPT = """cropped, hidden tail, extra limbs, extra ears, duplicate fox, other animals, blurry, text, watermark, photorealistic, 3d, plastic texture."""


def cargar_modelo():

    if not MODEL_DIR.exists():
        raise FileNotFoundError("No se encontro el modelo local.")

    print("\nCargando Stable Diffusion 1.5...")

    pipe = StableDiffusionPipeline.from_pretrained(
        str(MODEL_DIR),
        dtype=torch.float16,
        variant="fp16",
        use_safetensors=True,
        local_files_only=True,
        low_cpu_mem_usage=True
    )

    # Movemos las capas a GPU solamente cuando se necesitan.
    pipe.enable_sequential_cpu_offload()

    # Reducimos el consumo de memoria.
    # Usamos la atencion SDPA de PyTorch.
    pipe.vae.enable_slicing()

    pipe.set_progress_bar_config(desc="Generando imagen")

    return pipe


def generar_luma():

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

    print("=" * 50)
    print("FIFTHPLOT - GENERADOR DE IMAGENES")
    print("=" * 50)

    print(f"GPU: {torch.cuda.get_device_name(0)}")
    print("Precision: FP16")
    print(f"Resolucion: {IMAGE_SIZE}x{IMAGE_SIZE}")

    # Semilla fija para poder repetir el experimento.
    generador = torch.Generator(device="cpu").manual_seed(SEED)

    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()

    pipe = cargar_modelo()

    print("\nModelo cargado.")
    print("Generando ilustracion de Luma...")

    inicio = time.perf_counter()

    resultado = pipe(
        prompt=PROMPT,
        negative_prompt=NEGATIVE_PROMPT,
        height=IMAGE_SIZE,
        width=IMAGE_SIZE,
        num_inference_steps=STEPS,
        guidance_scale=GUIDANCE,
        generator=generador
    )

    tiempo = time.perf_counter() - inicio

    imagen = resultado.images[0]

    ruta_imagen = OUTPUT_DIR / "luma_referencia_v2.png"
    imagen.save(ruta_imagen)

    # Guardamos los parametros para documentar el experimento.
    memoria = 0

    if torch.cuda.is_available():
        memoria = round(
            torch.cuda.max_memory_allocated() / (1024 ** 3), 2
        )

    datos = {
        "modelo": "Stable Diffusion 1.5",
        "tipo_pesos": "FP16 Safetensors",
        "dispositivo": torch.cuda.get_device_name(0),
        "precision": "float16",
        "resolucion": f"{IMAGE_SIZE}x{IMAGE_SIZE}",
        "pasos": STEPS,
        "guidance_scale": GUIDANCE,
        "seed": SEED,
        "tiempo_segundos": round(tiempo, 2),
        "memoria_gpu_pico_gb": memoria,
        "prompt": PROMPT,
        "negative_prompt": NEGATIVE_PROMPT
    }

    ruta_datos = OUTPUT_DIR / "luma_referencia_v2.json"

    with open(ruta_datos, "w", encoding="utf-8") as archivo:
        json.dump(datos, archivo, ensure_ascii=False, indent=4)

    print("\nGeneracion terminada.")
    print(f"Imagen guardada: {ruta_imagen}")
    print(f"Parametros guardados: {ruta_datos}")
    print(f"Tiempo: {tiempo:.2f} segundos")
    print(f"Memoria GPU pico: {memoria} GB")

    # Liberamos la memoria para los siguientes modelos.
    del pipe
    gc.collect()
    torch.cuda.empty_cache()

    print("\nMemoria de GPU liberada.")


if __name__ == "__main__":

    if not torch.cuda.is_available():
        raise RuntimeError("No se detecto una GPU compatible con CUDA.")

    generar_luma()

import gc
import json
import time
import urllib.request
import urllib.error
from pathlib import Path

import torch
from PIL import Image, ImageDraw, ImageFont
from transformers import CLIPTokenizer

from fifthplot.generators import ip_adapter_test as ip_test


# 1. CONFIGURACION

MODEL = "qwen3:4b-q4_K_M"
OLLAMA_URL = "http://localhost:11434/api/chat"

STORY_DIR = Path("outputs/stories")
IMAGE_DIR = Path("outputs/images/final")
REFERENCE = Path("outputs/images/luma_prueba_01.png")
TOKENIZER_DIR = Path("models/stable-diffusion-v1-5/tokenizer")

IMAGE_SIZE = 512
STEPS = 20
GUIDANCE = 7.0
IP_SCALE = 0.5

BASE_SEED = 4200

ART_STYLE = "Watercolor storybook art, soft gouache brushstrokes, teal forest, warm colors."

LUMA_IDENTITY = "Luma, small red-orange fox, golden eyes, tiny nose, fluffy fur, large curled comet tail."

NEGATIVE_PROMPT = (
    "cropped, hidden tail, extra limbs, extra ears, "
    "duplicate fox, blurry, low quality, text, watermark, "
    "photorealistic, 3d render, plastic texture."
)

PROMPT_SCHEMA = {
    "type": "object",
    "properties": {
        "prompts": {
            "type": "array",
            "minItems": 5,
            "maxItems": 5,
            "items": {
                "type": "object",
                "properties": {
                    "number": {"type": "integer"},
                    "description_en": {"type": "string"}
                },
                "required": ["number", "description_en"]
            }
        }
    },
    "required": ["prompts"]
}


# 2. BUSCAR LA HISTORIA

def cargar_historia():

    historias = list(STORY_DIR.glob("*.json"))

    if not historias:
        raise FileNotFoundError(
            "No se encontraron historias en outputs/stories."
        )

    # Utilizamos la historia mas reciente.
    archivo = max(
        historias,
        key=lambda ruta: ruta.stat().st_mtime
    )

    with open(archivo, "r", encoding="utf-8") as f:
        historia = json.load(f)

    escenas = historia.get("scenes", [])

    if len(escenas) != 5:
        raise ValueError("La historia debe tener cinco escenas.")

    numeros = [escena["number"] for escena in escenas]

    if sorted(numeros) != [1, 2, 3, 4, 5]:
        raise ValueError("La numeracion de las escenas no es valida.")

    print(f"\nHistoria seleccionada: {historia['title']}")
    print(f"Archivo: {archivo}")

    return historia, archivo


# 3. TRADUCIR LOS PROMPTS CON QWEN3

def traducir_prompts(escenas):

    print("\nPreparando las descripciones visuales...")

    entradas = []

    for escena in escenas:
        entradas.append({
            "number": escena["number"],
            "title": escena["scene_title"],
            "description": escena["image_prompt"],
            "context": escena["narrative"]
        })

    mensaje = """
    Traduce al ingles las cinco descripciones visuales de un cuento.

    Devuelve exactamente cinco descripciones, numeradas del 1 al 5.

    Reglas:
    - Utiliza ingles sencillo para Stable Diffusion 1.5.
    - Cada descripcion debe tener como maximo 10 palabras.
    - Conserva la accion y el escenario de cada escena.
    - Incluye otros personajes cuando sean importantes.
    - No inventes acontecimientos.
    - No repitas la descripcion fisica de Luma.
    - No incluyas explicaciones ni texto adicional.

    DESCRIPCIONES:
    """ + json.dumps(entradas, ensure_ascii=False)

    datos = {
        "model": MODEL,
        "messages": [
            {
                "role": "system",
                "content": "You translate concise visual prompts for image generation."
            },
            {
                "role": "user",
                "content": mensaje
            }
        ],
        "format": PROMPT_SCHEMA,
        "think": False,
        "stream": False,
        "keep_alive": 0,
        "options": {
            "temperature": 0.2,
            "num_ctx": 4096,
            "num_predict": 1200
        }
    }

    solicitud = urllib.request.Request(
        OLLAMA_URL,
        data=json.dumps(datos).encode("utf-8"),
        headers={"Content-Type": "application/json"},
        method="POST"
    )

    try:
        with urllib.request.urlopen(solicitud, timeout=600) as respuesta:
            resultado = json.loads(respuesta.read().decode("utf-8"))

        contenido = resultado["message"]["content"]
        traducciones = json.loads(contenido)["prompts"]

    except urllib.error.URLError as error:
        raise RuntimeError(
            "No se pudo conectar con Ollama. "
            "Comprueba que Ollama este iniciado."
        ) from error

    numeros = [item["number"] for item in traducciones]

    if sorted(numeros) != [1, 2, 3, 4, 5]:
        raise ValueError("Qwen3 no devolvio las cinco traducciones.")

    traducciones.sort(key=lambda item: item["number"])

    print("Cinco descripciones traducidas correctamente.")

    return traducciones


# 4. PREPARAR LOS PROMPTS


def preparar_prompts(traducciones):

    tokenizer = CLIPTokenizer.from_pretrained(
        str(TOKENIZER_DIR),
        local_files_only=True
    )

    prompts = []

    for item in traducciones:

        descripcion = item["description_en"].strip()

        prefijo = f"{ART_STYLE} {LUMA_IDENTITY} Scene:"
        prompt = f"{prefijo} {descripcion}"

        tokens = tokenizer(
            prompt,
            truncation=False
        )["input_ids"]

        reducciones = 0

        # Reducimos solamente la descripcion de la escena.
        # Nunca eliminamos la identidad de Luma ni el estilo.
        while len(tokens) > 77:

            tokens_escena = tokenizer(
                descripcion,
                add_special_tokens=False
            )["input_ids"]

            if len(tokens_escena) <= 4:
                raise ValueError(
                    f"No se pudo ajustar el prompt de la escena "
                    f"{item['number']}."
                )

            descripcion = tokenizer.decode(
                tokens_escena[:-2],
                skip_special_tokens=True
            ).strip()

            prompt = f"{prefijo} {descripcion}"

            tokens = tokenizer(
                prompt,
                truncation=False
            )["input_ids"]

            reducciones += 1

        item["description_en"] = descripcion

        print(
            f"Escena {item['number']}: "
            f"{len(tokens)}/77 tokens"
        )

        if reducciones:
            print(
                f"Descripcion ajustada: "
                f"{reducciones} reducciones."
            )

        prompts.append(prompt)

    cantidad_negativo = len(
        tokenizer(
            NEGATIVE_PROMPT,
            truncation=False
        )["input_ids"]
    )

    print(f"Prompt negativo: {cantidad_negativo}/77 tokens")

    if cantidad_negativo > 77:
        raise ValueError(
            "El prompt negativo supera el limite de CLIP."
        )

    return prompts

# 6. CREAR HOJA DE CONTACTO

def crear_hoja_contacto(historia):

    nombres = [
        IMAGE_DIR / f"scene_{i:02d}.png"
        for i in range(1, 6)
    ]

    ancho = 360
    alto = 360
    margen = 20
    espacio = 20
    cabecera = 45

    lienzo = Image.new(
        "RGB",
        (margen * 2 + ancho * 2 + espacio,
         margen * 2 + (alto + cabecera) * 3),
        "white"
    )

    dibujo = ImageDraw.Draw(lienzo)

    for indice, ruta in enumerate(nombres):

        if not ruta.exists():
            continue

        imagen = Image.open(ruta).convert("RGB")
        imagen = imagen.resize((ancho, alto))

        columna = indice % 2
        fila = indice // 2

        x = margen + columna * (ancho + espacio)
        y = margen + fila * (alto + cabecera)

        dibujo.text(
            (x, y),
            f"Escena {indice + 1}",
            fill="black"
        )

        lienzo.paste(imagen, (x, y + cabecera))

        imagen.close()

    ruta_contacto = IMAGE_DIR / "hoja_contacto.png"

    lienzo.save(ruta_contacto)
    lienzo.close()

    print(f"\nHoja de contacto guardada: {ruta_contacto}")


# 7. EJECUCION PRINCIPAL

def main():

    historia, archivo_historia = cargar_historia()

    traducciones = traducir_prompts(historia["scenes"])

    prompts = preparar_prompts(traducciones)

    # Guardamos las traducciones antes de generar.
    IMAGE_DIR.mkdir(parents=True, exist_ok=True)

    with open(IMAGE_DIR / "prompts_traducidos.json", "w", encoding="utf-8") as f:
        json.dump(traducciones, f, ensure_ascii=False, indent=4)

    resultados = generar_ilustraciones(
        historia,
        archivo_historia,
        traducciones,
        prompts
    )

    crear_hoja_contacto(historia)

    tiempo_total = sum(
        item["tiempo_segundos"] for item in resultados
    )

    print("\n" + "=" * 55)
    print("ILUSTRACIONES GENERADAS CORRECTAMENTE")
    print("Escenas: 5 de 5")
    print(f"Tiempo total de generacion: {tiempo_total:.2f} segundos")
    print(f"Resultados: {IMAGE_DIR}")
    print("=" * 55)


if __name__ == "__main__":
    main()

import gc
import json
import time
from pathlib import Path

import torch
from PIL import Image
from diffusers import StableDiffusionPipeline
from transformers import CLIPTokenizer

# Rutas de los modelos y las imagenes.
BASE_DIR = Path("models/stable-diffusion-v1-5")
ADAPTER_DIR = Path("models/ip-adapter")
REFERENCE = Path("outputs/images/luma_referencia_v2.png")

OUTPUT_A = Path("experiments/configuration_a")
OUTPUT_B = Path("experiments/configuration_b")

# Parametros iguales para las dos pruebas.
SEED = 42
SIZE = 512
STEPS = 15
GUIDANCE = 7.0
ADAPTER_SCALE = 0.5

PROMPT = """Watercolor children's storybook illustration of Luma, the same small red-orange fox with golden eyes, tiny nose and fluffy fur, walking along a winding path in an enchanted teal forest at dusk. Full body, paws visible, large curled tail, soft gouache brushstrokes, warm fox against cool blue-green trees."""

NEGATIVE = """cropped, hidden tail, extra legs, extra ears, duplicate fox, blurry, text, watermark, photorealistic, 3d, plastic texture, multiple animals."""


def comprobar_archivos():

    archivos = [
        BASE_DIR / "model_index.json",
        ADAPTER_DIR / "models/ip-adapter-plus_sd15.safetensors",
        ADAPTER_DIR / "models/image_encoder/config.json",
        REFERENCE
    ]

    for archivo in archivos:
        if not archivo.exists():
            raise FileNotFoundError(f"No se encontro: {archivo}")

    # Revisamos que los prompts no superen el limite de CLIP.
    tokenizer = CLIPTokenizer.from_pretrained(
        str(BASE_DIR / "tokenizer"),
        local_files_only=True
    )

    for nombre, texto in [
        ("Positivo", PROMPT),
        ("Negativo", NEGATIVE)
    ]:
        cantidad = len(tokenizer(
            texto, truncation=False
        )["input_ids"])

        print(f"Prompt {nombre}: {cantidad}/77 tokens")

        if cantidad > 77:
            raise ValueError(f"El prompt {nombre} supera el limite.")


def cargar_pipeline(usando_referencia=False):

    print("\nCargando Stable Diffusion 1.5...")

    pipe = StableDiffusionPipeline.from_pretrained(
        str(BASE_DIR),
        dtype=torch.float16,
        variant="fp16",
        use_safetensors=True,
        local_files_only=True,
        low_cpu_mem_usage=True
    )

    pipe.vae.enable_slicing()

    if usando_referencia:

        print("Cargando IP-Adapter Plus...")

        pipe.load_ip_adapter(
            str(ADAPTER_DIR),
            subfolder="models",
            weight_name="ip-adapter-plus_sd15.safetensors",
            image_encoder_folder="image_encoder",
            local_files_only=True,
            low_cpu_mem_usage=True
        )

        pipe.set_ip_adapter_scale(ADAPTER_SCALE)

    # Movemos las capas a la GPU solo cuando se necesitan.
    pipe.enable_sequential_cpu_offload()

    return pipe


def generar_prueba(nombre, carpeta, usar_adapter=False):

    carpeta.mkdir(parents=True, exist_ok=True)

    pipe = cargar_pipeline(usar_adapter)

    referencia = None

    if usar_adapter:
        referencia = Image.open(REFERENCE).convert("RGB")

    generador = torch.Generator(device="cpu").manual_seed(SEED)

    torch.cuda.reset_peak_memory_stats()

    print(f"\nGenerando configuracion {nombre}...")
    inicio = time.perf_counter()

    parametros = {
        "prompt": PROMPT,
        "negative_prompt": NEGATIVE,
        "height": SIZE,
        "width": SIZE,
        "num_inference_steps": STEPS,
        "guidance_scale": GUIDANCE,
        "generator": generador
    }

    if usar_adapter:
        parametros["ip_adapter_image"] = referencia

    resultado = pipe(**parametros)

    tiempo = time.perf_counter() - inicio

    imagen = resultado.images[0]

    ruta = carpeta / f"luma_configuracion_{nombre}.png"
    imagen.save(ruta)

    memoria = round(
        torch.cuda.max_memory_allocated() / (1024 ** 3), 2
    )

    datos = {
        "configuracion": nombre,
        "modelo": "Stable Diffusion 1.5",
        "ip_adapter": usar_adapter,
        "referencia": str(REFERENCE) if usar_adapter else None,
        "adaptador": "IP-Adapter Plus SD15" if usar_adapter else None,
        "adapter_scale": ADAPTER_SCALE if usar_adapter else None,
        "seed": SEED,
        "resolucion": f"{SIZE}x{SIZE}",
        "pasos": STEPS,
        "guidance_scale": GUIDANCE,
        "tiempo_segundos": round(tiempo, 2),
        "memoria_gpu_pico_gb": memoria,
        "prompt": PROMPT,
        "negative_prompt": NEGATIVE
    }

    with open(carpeta / f"resultados_{nombre}.json", "w", encoding="utf-8") as archivo:
        json.dump(datos, archivo, ensure_ascii=False, indent=4)

    print(f"\nImagen guardada: {ruta}")
    print(f"Tiempo: {tiempo:.2f} segundos")
    print(f"Memoria GPU pico: {memoria} GB")

    del resultado
    del pipe

    gc.collect()
    torch.cuda.empty_cache()

    if torch.cuda.is_available():
        torch.cuda.ipc_collect()

    if referencia is not None:
        referencia.close()


def main():

    if not torch.cuda.is_available():
        raise RuntimeError("No se detecto la GPU.")

    comprobar_archivos()

    print("=" * 55)
    print("FIFTHPLOT - EXPERIMENTO DE CONSISTENCIA")
    print("=" * 55)

    # A: generacion solamente con texto.
    generar_prueba("A", OUTPUT_A, usar_adapter=False)

    # B: generacion con texto e imagen de referencia.
    generar_prueba("B", OUTPUT_B, usar_adapter=True)

    print("\nExperimento A/B finalizado.")
    print("Revisa las carpetas configuration_a y configuration_b.")


if __name__ == "__main__":
    main()

from pathlib import Path
from PIL import Image, ImageDraw
from rembg import new_session, remove

SOURCE = Path("outputs/images/luma_prueba_01.png")

OUTPUT_DIR = Path("outputs/assets/characters")

OUTPUT_IMAGE = OUTPUT_DIR / "luma_reference_rgba.png"
OUTPUT_PREVIEW = OUTPUT_DIR / "luma_reference_preview.png"
OUTPUT_MASK = OUTPUT_DIR / "luma_reference_mask.png"


def crear_fondo_cuadriculado(tamano):

    fondo = Image.new("RGB", tamano, "white")
    dibujo = ImageDraw.Draw(fondo)

    bloque = 32

    for y in range(0, tamano[1], bloque):
        for x in range(0, tamano[0], bloque):

            if (x // bloque + y // bloque) % 2 == 0:
                dibujo.rectangle(
                    (x, y, x + bloque, y + bloque),
                    fill=(215, 215, 215)
                )

    return fondo


def main():

    if not SOURCE.exists():
        raise FileNotFoundError(
            f"No se encontro la imagen original: {SOURCE}"
        )

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

    print("FIFTHPLOT - SEPARACION DE PERSONAJES")
    print("Imagen original:", SOURCE)

    # Utilizamos U2Net ejecutado en CPU.
    print("\nCargando modelo U2Net...")
    sesion = new_session(
        "u2net",
        providers=["CPUExecutionProvider"]
    )

    imagen = Image.open(SOURCE).convert("RGB")

    print("Eliminando el fondo...")
    resultado = remove(
        imagen,
        session=sesion,
        alpha_matting=False,
        post_process_mask=True
    )

    resultado = resultado.convert("RGBA")

    # Guardamos la imagen con transparencia real.
    resultado.save(OUTPUT_IMAGE)

    # Guardamos la mascara para revisar la segmentacion.
    mascara = resultado.getchannel("A")
    mascara.save(OUTPUT_MASK)

    # Creamos una vista previa con fondo cuadriculado.
    fondo = crear_fondo_cuadriculado(resultado.size).convert("RGBA")

    preview = Image.alpha_composite(
        fondo,
        resultado
    ).convert("RGB")

    preview.save(OUTPUT_PREVIEW)

    # Calculamos cuanto fondo quedo transparente.
    pixeles = list(mascara.getdata())

    transparentes = sum(
        1 for valor in pixeles if valor < 128
    )

    porcentaje = transparentes / len(pixeles) * 100

    print("\nProceso terminado.")
    print("PNG transparente:", OUTPUT_IMAGE)
    print("Vista previa:", OUTPUT_PREVIEW)
    print("Mascara:", OUTPUT_MASK)
    print(f"Area transparente: {porcentaje:.2f}%")

    imagen.close()
    resultado.close()
    mascara.close()
    fondo.close()
    preview.close()

    print("\nPersonaje procesado correctamente.")


if __name__ == "__main__":
    main()
